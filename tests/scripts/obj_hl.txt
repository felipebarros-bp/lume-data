dnode "syn A"

type "CREATE OR REPLACE FUNCTION lume_ui.calculate(p integer) RETURNS numeric LANGUAGE plpgsql AS $body$\nDECLARE msg text := 'a string\nthat spans lines';\nBEGIN /* outer /* nested */ still comment */\n  RETURN p * 3.0;\nEND;\n$body$"
wait 400
